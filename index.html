/* ===================================================== CAT SYSTEM ระบบแมวอย่างเดียว ===================================================== */ const catImage = document.getElementById("catImage"); /* ===================================================== CAT STATE ===================================================== */ /* ท่าปัจจุบันของแมว Sit = นั่ง / ปกติ Yawn = หาว Bored = เบื่อ Run_left = เดิน/วิ่งไปทางซ้าย Run_right = เดิน/วิ่งไปทางขวา Sleep = นอน Wet = เปียก / อาบน้ำ */ let catState = "Sit"; /* ตำแหน่งแมว ใช้เป็นเปอร์เซ็นต์ของหน้าจอ */ let catX = 50; let catY = 55; /* ทิศทางการเดิน 1 = ขวา -1 = ซ้าย */ let direction = 1; /* ===================================================== SET CAT POSE ===================================================== */ function setCatPose(pose){ catState = pose; catImage.src = pose; } /* ===================================================== CAT MOVEMENT ===================================================== */ function moveCat(){ /* ถ้าแมวกำลังนอน ห้ามเดิน */ if(catState === "Sleep"){ return; } /* ถ้าแมวเปียก ห้ามเดิน */ if(catState === "Wet"){ return; } /* ถ้าแมวกำลังเดินอยู่ ไม่ต้องเริ่มเดินซ้ำ */ if( catState === "Run_left" || catState === "Run_right" ){ return; } /* สุ่มว่าจะเดินไปทางไหน */ direction = Math.random() < 0.5 ? -1 : 1; /* เปลี่ยนท่าแมว */ if(direction === -1){ setCatPose("Run_left"); }else{ setCatPose("Run_right"); } /* สุ่มระยะทางที่จะเดิน */ const distance = 8 + Math.random() * 18; const startX = catX; let targetX = catX + (distance * direction); /* จำกัดไม่ให้แมวเดินออกนอกจอ */ targetX = Math.max( 10, Math.min(90, targetX) ); /* เวลาที่ใช้เดิน */ const duration = 1200 + Math.random() * 1000; const startTime = performance.now(); function walk(time){ const progress = Math.min( (time - startTime) / duration, 1 ); /* ทำให้การเดินนุ่มขึ้น */ const ease = progress < 0.5 ? 2 * progress * progress : 1 - Math.pow( -2 * progress + 2, 2 ) / 2; catX = startX + (targetX - startX) * ease; /* เปลี่ยนตำแหน่งแมว */ catImage.style.left = catX + "%"; /* เดินเสร็จแล้ว */ if(progress >= 1){ catX = targetX; catImage.style.left = catX + "%"; /* กลับมานั่งตามปกติ */ setCatPose("Sit"); return; } requestAnimationFrame(walk); } requestAnimationFrame(walk); } /* ===================================================== AUTOMATIC WALK ===================================================== */ /* แมวจะสุ่มเดินประมาณทุก 7–16 วินาที */ function scheduleNextWalk(){ const delay = 7000 + Math.random() * 9000; setTimeout(() => { moveCat(); scheduleNextWalk(); }, delay); } /* ===================================================== YAWN ===================================================== */ /* ทุก 1 นาทีแมวจะหาว แล้วกลับมานั่ง */ setInterval(() => { /* ไม่ขัดจังหวะตอนนอน ไม่ขัดจังหวะตอนเปียก ไม่ขัดจังหวะตอนเดิน */ if( catState === "Sleep" || catState === "Wet" || catState === "Run_left" || catState === "Run_right" ){ return; } setCatPose("Yawn"); /* หาวประมาณ 2 วินาที */ setTimeout(() => { if(catState === "Yawn"){ setCatPose("Sit"); } }, 2000); }, 60000); /* ===================================================== BORED ===================================================== */ /* ทุก 5 นาทีแมวจะแสดงท่า Bored แล้วกลับมานั่ง */ setInterval(() => { /* ไม่ขัดจังหวะท่าอื่น */ if( catState === "Sleep" || catState === "Wet" || catState === "Run_left" || catState === "Run_right" ){ return; } setCatPose("Bored"); /* เบื่อประมาณ 3 วินาที */ setTimeout(() => { if(catState === "Bored"){ setCatPose("Sit"); } }, 3000); }, 300000); /* ===================================================== START ===================================================== */ /* เริ่มต้นด้วยท่านั่ง */ setCatPose("Sit"); /* ตั้งตำแหน่งเริ่มต้น */ catImage.style.left = catX + "%"; catImage.style.top = catY + "%"; /* เริ่มระบบเดินอัตโนมัติ */ scheduleNextWalk();
